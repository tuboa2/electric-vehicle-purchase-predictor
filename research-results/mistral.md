from scipy.optimize import minimize

# --- LOGIT TRANSFORM ---
def logit(p):
    return np.log(p / (1 - p + 1e-15))

# --- COLLECT OOF PREDICTIONS (50 folds × 5 models = 250 predictions) ---

    logit(stream_a_oof), logit(stream_b_oof),
    logit(simpson_oof), logit(boundary_oof), logit(nn_oof)

def objective(weights):
    blended_logit = np.dot(oof_logits, weights)
